from django.core.exceptions import ObjectDoesNotExist, ValidationError
from accounts.models import EmployerProfile, EmployeeProfile, User

class ProfileManager:
    def create_employer_profile(self, user: User, data: dict) -> EmployerProfile:
        if hasattr(user, 'employer_profile'):
            raise ValidationError("Employer profile already exists.")
        return EmployerProfile.objects.create(user=user, **data)

    def update_employer_profile(self, user: User, data: dict) -> EmployerProfile:
        if not hasattr(user, 'employer_profile'):
            raise ValidationError("Employer profile does not exist.")
        profile = user.employer_profile
        for attr, value in data.items():
            setattr(profile, attr, value)
        profile.save()
        return profile

    def delete_employer_profile(self, user: User) -> None:
        if not hasattr(user, 'employer_profile'):
            raise ValidationError("Employer profile does not exist.")
        user.employer_profile.delete()

    def create_employee_profile(self, user: User, data: dict) -> EmployeeProfile:
        if hasattr(user, 'employee_profile'):
            raise ValidationError("Employee profile already exists.")
        return EmployeeProfile.objects.create(user=user, **data)

    def update_employee_profile(self, user: User, data: dict) -> EmployeeProfile:
        if not hasattr(user, 'employee_profile'):
            raise ValidationError("Employee profile does not exist.")
        profile = user.employee_profile
        for attr, value in data.items():
            setattr(profile, attr, value)
        profile.save()
        return profile

    def delete_employee_profile(self, user: User) -> None:
        if not hasattr(user, 'employee_profile'):
            raise ValidationError("Employee profile does not exist.")
        user.employee_profile.delete()

from rest_framework import viewsets, status
from rest_framework.permissions import IsAuthenticated
from rest_framework.response import Response
from rest_framework.decorators import action
from accounts.managers.profile_manager import ProfileManager
from accounts.serializers.profile_serializers import EmployerProfileSerializer, EmployeeProfileSerializer

class ProfileViewSet(viewsets.ViewSet):
    permission_classes = [IsAuthenticated]
    profile_manager = ProfileManager()

    @action(detail=False, methods=['post'], url_path='employer/create')
    def create_employer(self, request):
        serializer = EmployerProfileSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        profile = self.profile_manager.create_employer_profile(request.user, serializer.validated_data)
        return Response(EmployerProfileSerializer(profile).data, status=status.HTTP_201_CREATED)

    @action(detail=False, methods=['put'], url_path='employer/update')
    def update_employer(self, request):
        serializer = EmployerProfileSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        profile = self.profile_manager.update_employer_profile(request.user, serializer.validated_data)
        return Response(EmployerProfileSerializer(profile).data)

    @action(detail=False, methods=['delete'], url_path='employer/delete')
    def delete_employer(self, request):
        self.profile_manager.delete_employer_profile(request.user)
        return Response({"detail": "Employer profile deleted."}, status=status.HTTP_204_NO_CONTENT)

    @action(detail=False, methods=['post'], url_path='employee/create')
    def create_employee(self, request):
        serializer = EmployeeProfileSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        profile = self.profile_manager.create_employee_profile(request.user, serializer.validated_data)
        return Response(EmployeeProfileSerializer(profile).data, status=status.HTTP_201_CREATED)

    @action(detail=False, methods=['put'], url_path='employee/update')
    def update_employee(self, request):
        serializer = EmployeeProfileSerializer(data=request.data)
        serializer.is_valid(raise_exception=True)
        profile = self.profile_manager.update_employee_profile(request.user, serializer.validated_data)
        return Response(EmployeeProfileSerializer(profile).data)

    @action(detail=False, methods=['delete'], url_path='employee/delete')
    def delete_employee(self, request):
        self.profile_manager.delete_employee_profile(request.user)
        return Response({"detail": "Employee profile deleted."}, status=status.HTTP_204_NO_CONTENT)

# views.py
from rest_framework import viewsets, status
from rest_framework.response import Response
from rest_framework.permissions import IsAuthenticated
from accounts.models import EmployerProfile, EmployeeProfile
from accounts.serializers import EmployerProfileSerializer, EmployeeProfileSerializer
from accounts.permissions import IsEmployerUser, IsEmployeeUser

class EmployerProfileViewSet(viewsets.ViewSet):
    permission_classes = [IsAuthenticated, IsEmployerUser]

    def get_object(self, user):
        return EmployerProfile.objects.filter(user=user).first()

    def create(self, request):
        serializer = EmployerProfileSerializer(data=request.data)
        if serializer.is_valid():
            profile = EmployerProfile.objects.create_profile(user=request.user, **serializer.validated_data)
            return Response(EmployerProfileSerializer(profile).data, status=status.HTTP_201_CREATED)
        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

    def retrieve(self, request):
        profile = self.get_object(request.user)
        if not profile:
            return Response({"detail": "Profile not found"}, status=status.HTTP_404_NOT_FOUND)
        return Response(EmployerProfileSerializer(profile).data)

    def update(self, request):
        profile = self.get_object(request.user)
        if not profile:
            return Response({"detail": "Profile not found"}, status=status.HTTP_404_NOT_FOUND)
        serializer = EmployerProfileSerializer(profile, data=request.data, partial=True)
        if serializer.is_valid():
            profile = EmployerProfile.objects.update_profile(profile, **serializer.validated_data)
            return Response(EmployerProfileSerializer(profile).data)
        return Response(serializer.errors, status=status.HTTP_400_BAD_REQUEST)

    def destroy(self, request):
        profile = self.get_object(request.user)
        if profile:
            EmployerProfile.objects.delete_profile(profile)
            return Response(status=status.HTTP_204_NO_CONTENT)
        return Response({"detail": "Profile not found"}, status=status.HTTP_404_NOT_FOUND)
